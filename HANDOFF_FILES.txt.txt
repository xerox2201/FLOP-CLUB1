<script lang="ts">
  import { onMount, onDestroy } from 'svelte';
  import TopBar from '../components/TopBar.svelte';
  import { api } from '../lib/api';
  import Icon from '../components/Icon.svelte';

  let loading = true;
  let period: any = null;
  let entries: any[] = [];
  let myEntry: any = null;
  let myRank: number | null = null;
  let displayPrizePool = 0;
  let refreshTimer: any = null;
  let now = Date.now();
  let clockTimer: any = null;

  function fmt(n: any): string {
    return Number(Number(n || 0).toFixed(2)).toFixed(2);
  }
  function handsLabel(n: number): string {
    return n === 1 ? 'hand' : 'hands';
  }
  function normalize(e: any, idx: number) {
    return {
      user_id: e.user_id ?? e.userId ?? e.id,
      rank: e.rank ?? idx + 1,
      avatar_id: e.avatar_id ?? e.avatarId ?? 'spade',
      display_name: e.display_name ?? e.name ?? 'Player',
      hands_played: e.hands_played ?? 0,
      total_won_usd: e.total_won_usd ?? 0,
      prize_usd: e.prize_usd ?? 0,
    };
  }

  async function loadAll() {
    try {
      const [cur, me] = await Promise.all([
        api.request<any>('/api/leaderboard/current'),
        api.request<any>('/api/leaderboard/me'),
      ]);
      if (cur.ok && cur.data) {
        period = cur.data.period ?? null;
        const raw = cur.data.entries ?? cur.data.leaderboard ?? [];
        entries = raw.map((e: any, i: number) => normalize(e, i));
        displayPrizePool = cur.data.livePrizePool ?? cur.data.prizePool ?? 0;
      }
      if (me.ok && me.data) {
        myEntry = me.data.entry ? normalize(me.data.entry, 0) : null;
        myRank = me.data.liveRank ?? me.data.rank ?? null;
      }
    } catch (e) {
      console.error('[Leaderboard] load failed', e);
    } finally {
      loading = false;
    }
  }

  async function refresh() {
    try {
      const res = await api.request<any>('/api/leaderboard/current');
      if (res.ok && res.data) {
        const raw = res.data.entries ?? res.data.leaderboard ?? [];
        entries = raw.map((e: any, i: number) => normalize(e, i));
        period = res.data.period ?? null;
        displayPrizePool = res.data.livePrizePool ?? res.data.prizePool ?? 0;
      }
    } catch {}
  }

  onMount(async () => {
    await loadAll();
    refreshTimer = setInterval(refresh, 5000);
    clockTimer = setInterval(() => {
      now = Date.now();
    }, 1000);
  });

  onDestroy(() => {
    if (refreshTimer) clearInterval(refreshTimer);
    if (clockTimer) clearInterval(clockTimer);
  });

  // Countdown to period end
  $: endAt = period?.period_end ? new Date(period.period_end).getTime() : 0;
  $: remainingMs = Math.max(0, endAt - now);
  $: countdown = (() => {
    if (remainingMs <= 0) return 'Ended';
    const s = Math.floor(remainingMs / 1000);
    const d = Math.floor(s / 86400);
    const h = Math.floor((s % 86400) / 3600);
    const m = Math.floor((s % 3600) / 60);
    const sec = s % 60;
    if (d > 0) return `${d}d ${h}h ${m}m`;
    if (h > 0) return `${h}h ${m}m ${sec}s`;
    if (m > 0) return `${m}m ${sec}s`;
    return `${sec}s`;
  })();

  // ─── Prize distribution ───
  const PRIZE_DISTRIBUTION = [
    { from: 1, to: 1, pct: 0.21, label: '1st' },
    { from: 2, to: 2, pct: 0.12, label: '2nd' },
    { from: 3, to: 3, pct: 0.08, label: '3rd' },
    { from: 4, to: 4, pct: 0.06, label: '4th' },
    { from: 5, to: 5, pct: 0.05, label: '5th' },
    { from: 6, to: 10, pct: 0.04, label: '6-10' },
    { from: 11, to: 15, pct: 0.025, label: '11-15' },
    { from: 16, to: 20, pct: 0.015, label: '16-20' },
    { from: 21, to: 25, pct: 0.01, label: '21-25' },
    { from: 26, to: 30, pct: 0.006, label: '26-30' },
  ];

  function pctForRank(rank: number): number {
    const tier = PRIZE_DISTRIBUTION.find((t) => rank >= t.from && rank <= t.to);
    return tier ? tier.pct : 0;
  }

  function prizeForRank(rank: number, pool: number): number {
    const pct = pctForRank(rank);
    if (pct === 0) return 0;
    return Math.round(pool * pct * 100) / 100;
  }

  // Season name
  $: seasonName = (() => {
    if (!period?.period_start) return 'Season';
    try {
      const d = new Date(period.period_start);
      return d.toLocaleDateString('en-US', { month: 'long', year: 'numeric' });
    } catch {
      return 'Season';
    }
  })();

  $: first = entries[0] ?? null;
  $: second = entries[1] ?? null;
  $: third = entries[2] ?? null;
  $: rest = entries.slice(3);
</script>

import type { FastifyInstance } from 'fastify';

const JWT_SECRET = process.env.JWT_SECRET ?? 'dev_secret_change_me';

function authUser(req: any): number | null {
  const auth = req.headers.authorization;
  if (!auth?.startsWith('Bearer ')) return null;
  const payload = verifyToken(auth.slice(7), JWT_SECRET);
  return payload?.uid ?? null;
}

export async function registerHomeRoutes(app: FastifyInstance, db: Pool) {
  app.get('/api/home', async (req) => {
    const userId = authUser(req);

    // ─── 1. Live stats from in-memory gameManager ───
    const rooms = gameManager.getRooms();
    const cashRooms = rooms.filter((r: any) => !r.tournamentId && r.players.size > 0);
    const tourneyRooms = rooms.filter((r: any) => r.tournamentId && r.players.size > 0);

    // ─── 2. Live cash tables list (from DB meta + memory players) ───
    let liveTables: any[] = [];
    if (cashRooms.length > 0) {
      const ids = cashRooms.map((r: any) => r.tableId);
      const meta = await db.query(
        `SELECT id, name, variant, mode, blinds, seats FROM poker_tables WHERE id = ANY($1::bigint[])`,
        [ids],
      );
      const metaMap = new Map(meta.rows.map((r: any) => [Number(r.id), r]));
      liveTables = cashRooms
        .map((r: any) => {
          const m: any = metaMap.get(r.tableId);
          if (!m) return null;
          return {
            id: r.tableId,
            name: m.name,
            variant: m.variant,
            mode: m.mode,
            blinds: {
              sb: Number(m.blinds?.sb ?? 0),
              bb: Number(m.blinds?.bb ?? 0),
            },
            seats: Number(m.seats ?? 6),
            players: r.players.size,
            phase: r.game?.state?.phase ?? 'waiting',
          };
        })
        .filter(Boolean)
        .slice(0, 4);
    }

    // ─── 3. Live tournaments with prize > 0 ───
    const tourRes = await db.query(
      `SELECT t.id, t.name, t.variant, t.speed, t.total_prize_pool, t.current_level,
              (SELECT COUNT(*)::int FROM tournament_entries
               WHERE tournament_id = t.id AND status IN ('registered','playing')) AS players
       FROM tournaments t
       WHERE t.status IN ('running','final_table')
         AND t.total_prize_pool > 0
       ORDER BY t.total_prize_pool DESC
       LIMIT 4`,
    );
    const liveTournaments = tourRes.rows.map((r: any) => ({
      id: Number(r.id),
      name: r.name,
      variant: r.variant,
      speed: r.speed,
      totalPrizePool: Number(r.total_prize_pool),
      players: Number(r.players),
      currentLevel: Number(r.current_level),
    }));

    // ─── 4. Top winners today (real money) ───
    const topWinnersToday = await getTopWinnersToday(db);

    // ─── 5. User progress (only if authed) ───
    let userProgress: any = null;
    if (userId) {
      try {
        const { LoyaltyService } = await import('./loyalty/service.js');
        const svc = new LoyaltyService(db);
        const loyalty = await svc.getSummary(userId);
        userProgress = {
          level: loyalty.level,
          xpIntoLevel: loyalty.xpIntoLevel,
          xpForNextLevel: loyalty.xpForNextLevel,
          tier: loyalty.tier,
          tierLabel: loyalty.tierLabel,
          tierIcon: loyalty.tierIcon,
          tierColor: loyalty.tierColor,
          rakebackPct: loyalty.rakebackPct,
          progressToNextTier: loyalty.progressToNextTier,
        };
      } catch (e: any) {
        console.error('[home] userProgress failed:', e.message);
      }
    }

    return {
      onlineUsers: getOnlineUserCount(),
      activeTables: cashRooms.length,
      activeTournaments: tourneyRooms.length,
      tables: liveTables,
      tournaments: liveTournaments,
      topWinnersToday,
      userProgress,
    };
  });
}

// Aggregate top 3 winners from the last 24h of real-money hands
// ============================================================
async function getTopWinnersToday(db: Pool): Promise<any[]> {
  try {
    const res = await db.query(
      `SELECT h.result, hs.state
       FROM hands h
       JOIN hand_snapshots hs ON hs.hand_id = h.id AND hs.sequence = 0
       WHERE h.mode = 'real'
         AND h.finished = TRUE
         AND h.ended_at > NOW() - INTERVAL '24 hours'
       ORDER BY h.id DESC
       LIMIT 500`,
    );

    const winnersMap = new Map<
      number,
      { userId: number; displayName: string; totalWon: number; wins: number }
    >();

    for (const row of res.rows) {
      const state = row.state || {};
      const players = state.players || [];
      const showdown = row.result?.showdownPlayers || [];

      for (const sp of showdown) {
        if (!sp.isWinner) continue;
        const payout = Number(sp.payout) || 0;
        if (payout <= 0) continue;

        const player = players.find((p: any) => Number(p.seat) === Number(sp.seat));
        if (!player) continue;

        const uid = Number(player.userId);
        if (!uid) continue;

        if (!winnersMap.has(uid)) {
          winnersMap.set(uid, {
            userId: uid,
            displayName: player.displayName || 'Player',
            totalWon: 0,
            wins: 0,
          });
        }
        const entry = winnersMap.get(uid)!;
        entry.totalWon += payout;
        entry.wins += 1;
      }
    }

    const sorted = [...winnersMap.values()].sort((a, b) => b.totalWon - a.totalWon).slice(0, 3);

    if (sorted.length > 0) {
      const ids = sorted.map((s) => s.userId);
      const avatarRes = await db.query(
        `SELECT id, avatar_id FROM users WHERE id = ANY($1::bigint[])`,
        [ids],
      );
      const avatarMap = new Map(avatarRes.rows.map((r: any) => [Number(r.id), r.avatar_id]));
      for (const s of sorted) {
        (s as any).avatarId = avatarMap.get(s.userId) || 'king';
      }
    }

    return sorted;
  } catch (e: any) {
    console.error('[home] topWinnersToday failed:', e.message);
    return [];
  }
}

export function tierForRake(lifetimeRake: number): TierDef {
  const r = Math.max(0, Number(lifetimeRake) || 0);
  for (const t of TIERS) {
    if (r >= t.minRake && r < t.maxRake) return t;
  }
  return TIERS[TIERS.length - 1];
}

export function nextTier(current: TierName): TierDef | null {
  const idx = TIERS.findIndex((t) => t.name === current);
  if (idx < 0 || idx >= TIERS.length - 1) return null;
  return TIERS[idx + 1];
}

export function tierDef(name: TierName): TierDef {
  return TIERS.find((t) => t.name === name) ?? TIERS[0];
}

// ============================================================
// XP CURVE — stepped, exact table
// ============================================================
// XP_TABLE[L] = XP needed to go from level L to L+1
// Covers L1..L49. For L50+ computed separately.
const XP_TABLE: number[] = [
  0,          // index 0 unused
  // Tier I — Starter (L1-L5 entry points: 4 transitions)
  100, 250, 500, 1000,
  // Tier II — Bronze (L5→L15, 10 transitions)
  1500, 2200, 3000, 4000, 5500, 7000, 9000, 12000, 16000, 20000,
  // Tier III — Silver (L15→L30, 15 transitions)
  30000, 40000, 50000, 60000, 75000, 100000, 130000, 165000, 210000, 260000,
  320000, 400000, 500000, 620000, 770000,
  // Tier IV — Gold (L30→L50, 20 transitions)
  1000000, 1200000, 1450000, 1750000, 2100000, 2500000, 3000000, 3600000, 4300000, 5200000,
  6200000, 7500000, 9000000, 10800000, 13000000, 15600000, 18700000, 22500000, 27000000, 32500000,
];

export function xpForNextLevel(level: number): number {
  if (level < 1) return 100;

  // Tier I-III + IV (L1-L49): exact table
  if (level <= 49) {
    return XP_TABLE[level] ?? 32_500_000;
  }

  // Tier V — Platinum (L50-L79): anchor-based linear interpolation
  //   L50→L51 = 40M, L60→L61 = 150M, L70→L71 = 350M, L79→L80 = 500M
  if (level <= 59) return Math.round(40_000_000 + (level - 50) * 11_000_000);
  if (level <= 69) return Math.round(150_000_000 + (level - 60) * 20_000_000);
  if (level <= 79) return Math.round(350_000_000 + (level - 70) * (150_000_000 / 9));

  // Tier VI — Diamond (L80+): 500M + 50M per level, cap at 1B
  const xp = 500_000_000 + (level - 80) * 50_000_000;
  return Math.min(1_000_000_000, xp);
}

export interface LevelInfo {
  level: number;
  xpIntoLevel: number;
  xpForNextLevel: number;
  xpAtLevelStart: number;
}

export function levelFromLifetimeXP(lifetimeXp: number): LevelInfo {
  let remaining = Math.max(0, Number(lifetimeXp) || 0);
  let level = 1;
  let consumed = 0;

  for (let safety = 0; safety < 5000; safety++) {
    const need = xpForNextLevel(level);
    if (remaining < need) {
      return { level, xpIntoLevel: remaining, xpForNextLevel: need, xpAtLevelStart: consumed };
    }
    remaining -= need;
    consumed += need;
    level++;
  }
  return { level, xpIntoLevel: 0, xpForNextLevel: xpForNextLevel(level), xpAtLevelStart: consumed };
}

  if (level >= 10) return 0.05;
  return 0;
}

// ============================================================
// LoyaltyService — pays rakeback + XP per hand,
//                   updates tiers + voucher progress,
//                   exposes read APIs for /api/me + routes.
// ============================================================

import { Pool } from 'pg';
import { LedgerService } from '../ledger.js';
import {
  TierName,
  tierForRake,
  nextTier,
  XP_POOL_PCT,
  XP_PER_DOLLAR,
  SEED_DEFS,
  SeedType,
  seedDef,
  levelFromLifetimeXP,
} from './definitions.js';

export interface PlayerContribution {
  userId: number;
  contribution: number;
}

export interface PayHandResult {
  handId: number;
  totalRake: number;
  pot: number;
  perPlayer: Array<{
    userId: number;
    rakeShare: number;
    rakeback: number;
    xp: number;
    newTier: TierName;
    skipped: boolean;
    reason?: string;
  }>;
}

export class LoyaltyService {
  private ledger: LedgerService;

  constructor(private db: Pool) {
    this.ledger = new LedgerService(db);
  }

  // ============================================================
  // payHand — pay rakeback + XP for a single hand (idempotent)
  // ============================================================
  async payHand(params: {
    handId: number;
    pot: number;
    totalRake: number;
    contributions: PlayerContribution[];
  }): Promise<PayHandResult> {
    const { handId, pot, totalRake, contributions } = params;
    const result: PayHandResult = { handId, totalRake, pot, perPlayer: [] };

    if (totalRake <= 0 || pot <= 0) return result;

    for (const c of contributions) {
      if (c.contribution <= 0) {
        result.perPlayer.push({
          userId: c.userId,
          rakeShare: 0,
          rakeback: 0,
          xp: 0,
          newTier: 'welcome',
          skipped: true,
          reason: 'no_contribution',
        });
        continue;
      }

      const share = c.contribution / pot;
      const rakeShare = Math.round(totalRake * share * 100) / 100;
      if (rakeShare <= 0) {
        result.perPlayer.push({
          userId: c.userId,
          rakeShare: 0,
          rakeback: 0,
          xp: 0,
          newTier: 'welcome',
          skipped: true,
          reason: 'rake_share_zero',
        });
        continue;
      }

      // Idempotency
      const existing = await this.db.query(
        'SELECT id FROM rakeback_log WHERE hand_id = $1 AND user_id = $2',
        [handId, c.userId],
      );
      if (existing.rows.length > 0) {
        const tierRow = await this.db.query(
          'SELECT tier FROM user_tier WHERE user_id = $1',
          [c.userId],
        );
        result.perPlayer.push({
          userId: c.userId,
          rakeShare,
          rakeback: 0,
          xp: 0,
          newTier: (tierRow.rows[0]?.tier as TierName) ?? 'welcome',
          skipped: true,
          reason: 'already_paid',
        });
        continue;
      }

      // Current tier
      const tierRow = await this.db.query(
        'SELECT tier, lifetime_rake FROM user_tier WHERE user_id = $1',
        [c.userId],
      );
      const currentTier: TierName =
        (tierRow.rows[0]?.tier as TierName) ?? 'welcome';
      const lifetimeRakeBefore = Number(tierRow.rows[0]?.lifetime_rake ?? 0);

      const tierDef = tierForRake(lifetimeRakeBefore);
      const rakeback = Math.round(rakeShare * tierDef.rakebackPct * 100) / 100;
      const xp = Math.round(rakeShare * XP_POOL_PCT * XP_PER_DOLLAR);

      const client = await this.db.connect();
      try {
        await client.query('BEGIN');

        // 1. Credit rakeback via ledger
        if (rakeback > 0) {
          const userAcc = await this.ledger.getUserAccountId(c.userId);
          const houseAcc = await this.ledger.getAccountId('HOUSE_RAKE');
          if (userAcc && houseAcc) {
            await this.ledger.record({
              kind: 'BONUS',
              userId: c.userId,
              ref: `rakeback_hand_${handId}`,
              meta: { handId, rakeShare, rakebackPct: tierDef.rakebackPct },
              entries: [
                { accountId: houseAcc, amount: -rakeback },
                { accountId: userAcc, amount: rakeback },
              ],
              idempotencyKey: `rakeback_hand_${handId}_user_${c.userId}`,
            });
          }
        }

        // 2. Log rakeback
        await client.query(
          `INSERT INTO rakeback_log (hand_id, user_id, amount)
           VALUES ($1, $2, $3)
           ON CONFLICT (hand_id, user_id) DO NOTHING`,
          [handId, c.userId, rakeback],
        );

        // 3. Credit XP
        if (xp > 0) {
          await client.query(
            `INSERT INTO user_xp (user_id, lifetime_xp, claimed_xp, updated_at)
             VALUES ($1, $2, 0, NOW())
             ON CONFLICT (user_id) DO UPDATE
               SET lifetime_xp = user_xp.lifetime_xp + EXCLUDED.lifetime_xp,
                   updated_at = NOW()`,
            [c.userId, xp],
          );
          await client.query(
            `INSERT INTO xp_transactions (user_id, amount, source, ref_id, meta)
             VALUES ($1, $2, 'hand_rakeback', $3, $4)`,
            [c.userId, xp, handId, JSON.stringify({ rakeShare })],
          );
        }

        // 4. Update lifetime_rake + tier
        const newLifetimeRake =
          Math.round((lifetimeRakeBefore + rakeShare) * 100) / 100;
        const newTierDef = tierForRake(newLifetimeRake);
        await client.query(
          `INSERT INTO user_tier (user_id, tier, lifetime_rake, updated_at)
           VALUES ($1, $2, $3, NOW())
           ON CONFLICT (user_id) DO UPDATE
             SET tier = EXCLUDED.tier,
                 lifetime_rake = EXCLUDED.lifetime_rake,
                 updated_at = NOW()`,
          [c.userId, newTierDef.name, newLifetimeRake],
        );

        // 4b. Decrement bonus pending counters
        await client.query(
          `UPDATE users
           SET bonus_playthrough_pending = GREATEST(0, bonus_playthrough_pending - $1),
               bonus_rake_pending = GREATEST(0, bonus_rake_pending - $2)
           WHERE id = $3`,
          [c.contribution, rakeShare, c.userId],
        );

        // 5. Voucher progress
        await client.query(
          `UPDATE vouchers
           SET current_rake = current_rake + $1
           WHERE user_id = $2
             AND used = FALSE
             AND expires_at > NOW()
             AND current_rake < min_rake`,
          [rakeShare, c.userId],
        );

        // 6. Auto-unlock vouchers
        const unlocked = await client.query(
          `SELECT id, amount FROM vouchers
           WHERE user_id = $1
             AND used = FALSE
             AND expires_at > NOW()
             AND current_rake >= min_rake
           RETURNING id, amount`,
          [c.userId],
        );
        for (const v of unlocked.rows) {
          const userAcc = await this.ledger.getUserAccountId(c.userId);
          const bonusAcc = await this.ledger.getAccountId('HOUSE_BONUS');
          const amt = Number(v.amount);
          if (userAcc && bonusAcc) {
            const ok = await this.ledger.record({
              kind: 'BONUS',
              userId: c.userId,
              ref: `voucher_unlock_${v.id}`,
              meta: { voucherId: v.id },
              entries: [
                { accountId: bonusAcc, amount: -amt },
                { accountId: userAcc, amount: amt },
              ],
              idempotencyKey: `voucher_unlock_${v.id}`,
            });
            if (ok.ok) {
              await client.query(
                `UPDATE vouchers SET used = TRUE, used_at = NOW() WHERE id = $1`,
                [v.id],
              );
              await client.query(
                `UPDATE users
                 SET bonus_playthrough_pending = bonus_playthrough_pending + $1,
                     bonus_rake_pending = bonus_rake_pending + 15,
                     bonus_real_hands_pending = bonus_real_hands_pending + 150,
                     bonus_received_at = NOW()
                 WHERE id = $2`,
                [amt * 10, c.userId],
              );
            }
          }
        }

        await client.query('COMMIT');

        result.perPlayer.push({
          userId: c.userId,
          rakeShare,
          rakeback,
          xp,
          newTier: newTierDef.name,
          skipped: false,
        });
      } catch (e: any) {
        await client.query('ROLLBACK');
        console.error(
          `[loyalty] payHand user=${c.userId} hand=${handId} failed:`,
          e.message,
        );
        result.perPlayer.push({
          userId: c.userId,
          rakeShare,
          rakeback: 0,
          xp: 0,
          newTier: currentTier,
          skipped: true,
          reason: 'error:' + e.message,
        });
      } finally {
        client.release();
      }
    }

    return result;
  }

  // ============================================================
  // getSummary — everything for /api/me loyalty block
  // ============================================================
  async getSummary(userId: number) {
    const xpRow = await this.db.query(
      'SELECT lifetime_xp, claimed_xp FROM user_xp WHERE user_id = $1',
      [userId],
    );
    const lifetimeXp = Number(xpRow.rows[0]?.lifetime_xp ?? 0);
    const claimedXp = Number(xpRow.rows[0]?.claimed_xp ?? 0);
    const availableXp = Math.max(0, lifetimeXp - claimedXp);
    const levelInfo = levelFromLifetimeXP(lifetimeXp);

    const tierRow = await this.db.query(
      'SELECT tier, lifetime_rake FROM user_tier WHERE user_id = $1',
      [userId],
    );
    const lifetimeRake = Number(tierRow.rows[0]?.lifetime_rake ?? 0);
    const tierDef = tierForRake(lifetimeRake);
    const nt = nextTier(tierDef.name);

    let progressToNextTier = 1;
    if (nt) {
      const span = nt.minRake - tierDef.minRake;
      progressToNextTier =
        span > 0
          ? Math.min(1, Math.max(0, (lifetimeRake - tierDef.minRake) / span))
          : 0;
    }

    const rbRow = await this.db.query(
      'SELECT COALESCE(SUM(amount), 0)::numeric AS total FROM rakeback_log WHERE user_id = $1',
      [userId],
    );
    const rakebackEarned = Number(rbRow.rows[0]?.total ?? 0);

    const vRows = await this.db.query(
      `SELECT id, amount, min_rake, current_rake, used, expires_at
       FROM vouchers
       WHERE user_id = $1
       ORDER BY created_at DESC
       LIMIT 20`,
      [userId],
    );
    const vouchers = vRows.rows.map((r: any) => {
      const minRake = Number(r.min_rake);
      const current = Number(r.current_rake);
      return {
        id: Number(r.id),
        amount: Number(r.amount),
        minRake,
        currentRake: current,
        used: !!r.used,
        expiresAt: r.expires_at,
        progressPct:
          minRake > 0
            ? Math.min(100, Math.round((current / minRake) * 100))
            : 0,
      };
    });

    const fc = await this.db.query(
      'SELECT total_hands, free_hands_won FROM free_chips WHERE user_id = $1',
      [userId],
    );
    const freeHands = Number(fc.rows[0]?.total_hands ?? 0);
    const freeWins = Number(fc.rows[0]?.free_hands_won ?? 0);

    const sRows = await this.db.query(
      'SELECT seed_type, claimed, claimed_at FROM seeds WHERE user_id = $1',
      [userId],
    );
    const claimedMap = new Map<
      string,
      { claimed: boolean; at: string | null }
    >();
    for (const r of sRows.rows) {
      claimedMap.set(r.seed_type, { claimed: !!r.claimed, at: r.claimed_at });
    }

    const seeds = SEED_DEFS.map((def) => {
      const claimed = claimedMap.get(def.type);
      const eligible =
        !claimed?.claimed &&
        freeHands >= def.minFreeHands &&
        freeWins >= def.minFreeWins;
      return {
        type: def.type,
        label: def.label,
        icon: def.icon,
        amount: def.amount,
        claimed: !!claimed?.claimed,
        claimedAt: claimed?.at ?? null,
        eligible,
        progress: {
          freeHands,
          freeWins,
          minFreeHands: def.minFreeHands,
          minFreeWins: def.minFreeWins,
        },
      };
    });

    return {
      lifetimeXp,
      claimedXp,
      availableXp,
      level: levelInfo.level,
      xpIntoLevel: levelInfo.xpIntoLevel,
      xpForNextLevel: levelInfo.xpForNextLevel,
      tier: tierDef.name,
      tierLabel: tierDef.label,
      tierIcon: tierDef.icon,
      tierColor: tierDef.color,
      lifetimeRake,
      rakebackPct: tierDef.rakebackPct,
      nextTier: nt
        ? { name: nt.name, label: nt.label, minRake: nt.minRake }
        : null,
      progressToNextTier,
      rakePaidLifetime: lifetimeRake,
      rakebackEarned,
      vouchers,
      seeds,
    };
  }

  // ============================================================
  // claimSeed
  // ============================================================
  async claimSeed(
    userId: number,
    type: SeedType,
  ): Promise<{ ok: boolean; error?: string; amount?: number }> {
    const def = seedDef(type);

    const existing = await this.db.query(
      'SELECT claimed FROM seeds WHERE user_id = $1 AND seed_type = $2',
      [userId, type],
    );
    if (existing.rows[0]?.claimed)
      return { ok: false, error: 'already_claimed' };

    const fc = await this.db.query(
      'SELECT total_hands, free_hands_won FROM free_chips WHERE user_id = $1',
      [userId],
    );
    const freeHands = Number(fc.rows[0]?.total_hands ?? 0);
    const freeWins = Number(fc.rows[0]?.free_hands_won ?? 0);

    if (freeHands < def.minFreeHands)
      return { ok: false, error: `need_${def.minFreeHands}_free_hands` };
    if (freeWins < def.minFreeWins)
      return { ok: false, error: `need_${def.minFreeWins}_free_wins` };

    const userAcc = await this.ledger.getUserAccountId(userId);
    const bonusAcc = await this.ledger.getAccountId('HOUSE_BONUS');
    if (!userAcc || !bonusAcc) return { ok: false, error: 'account_missing' };

    const ok = await this.ledger.record({
      kind: 'BONUS',
      userId,
      ref: `seed_${type}`,
      meta: { seedType: type, amount: def.amount },
      entries: [
        { accountId: bonusAcc, amount: -def.amount },
        { accountId: userAcc, amount: def.amount },
      ],
      idempotencyKey: `seed_${type}_user_${userId}`,
    });

    if (!ok.ok) return { ok: false, error: ok.error ?? 'ledger_failed' };

    await this.db.query(
      `INSERT INTO seeds (user_id, seed_type, amount, claimed, claimed_at)
       VALUES ($1, $2, $3, TRUE, NOW())
       ON CONFLICT (user_id, seed_type) DO UPDATE
         SET claimed = TRUE, claimed_at = NOW()`,
      [userId, type, def.amount],
    );

    await this.db.query(
      `UPDATE users
       SET bonus_playthrough_pending = bonus_playthrough_pending + $1,
           bonus_rake_pending = bonus_rake_pending + $2,
           bonus_real_hands_pending = bonus_real_hands_pending + $3,
           bonus_received_at = NOW()
       WHERE id = $4`,
      [
        def.amount * def.playthroughMultiplier,
        def.minRakeAfter,
        def.minRealHandsAfter,
        userId,
      ],
    );

    return { ok: true, amount: def.amount };
  }

  // ============================================================
  // cashOutXp — with level gate + 24h cooldown + monthly cap
  // ============================================================
  async cashOutXp(
    userId: number,
    xpAmount: number,
  ): Promise<{
    ok: boolean;
    error?: string;
    usd?: number;
    newClaimedXp?: number;
  }> {
    const xp = Math.floor(xpAmount);
    if (xp < 5000) return { ok: false, error: 'min_5000_xp' };
    if (xp % 1000 !== 0)
      return { ok: false, error: 'must_be_multiple_of_1000' };

    const xpRow = await this.db.query(
      'SELECT lifetime_xp, claimed_xp FROM user_xp WHERE user_id = $1',
      [userId],
    );
    const lifetimeXp = Number(xpRow.rows[0]?.lifetime_xp ?? 0);
    const claimedXp = Number(xpRow.rows[0]?.claimed_xp ?? 0);
    const available = Math.max(0, lifetimeXp - claimedXp);

    const lvl = levelFromLifetimeXP(lifetimeXp).level;
    if (lvl < 5) return { ok: false, error: 'level_5_required' };

    if (available < xp) return { ok: false, error: 'insufficient_xp' };

    const recent = await this.db.query(
      `SELECT created_at FROM xp_transactions
       WHERE user_id = $1 AND source = 'claim'
       ORDER BY created_at DESC LIMIT 1`,
      [userId],
    );
    if (recent.rows[0]) {
      const last = new Date(recent.rows[0].created_at).getTime();
      const elapsedHours = (Date.now() - last) / 3_600_000;
      if (elapsedHours < 24) {
        const waitH = Math.ceil(24 - elapsedHours);
        return { ok: false, error: `cooldown_${waitH}h` };
      }
    }

    const monthly = await this.db.query(
      `SELECT COALESCE(SUM(ABS(amount)), 0)::bigint AS total
       FROM xp_transactions
       WHERE user_id = $1 AND source = 'claim'
         AND created_at > NOW() - INTERVAL '30 days'`,
      [userId],
    );
    const monthTotal = Number(monthly.rows[0]?.total ?? 0);
    if (monthTotal + xp > 50000) {
      return {
        ok: false,
        error: `monthly_cap_50000_xp (used ${monthTotal})`,
      };
    }

    const usd = xp / 1000;
    const userAcc = await this.ledger.getUserAccountId(userId);
    const bonusAcc = await this.ledger.getAccountId('HOUSE_BONUS');
    if (!userAcc || !bonusAcc) return { ok: false, error: 'account_missing' };

    const client = await this.db.connect();
    try {
      await client.query('BEGIN');

      const ok = await this.ledger.record({
        kind: 'BONUS',
        userId,
        ref: `xp_cashout_${xp}`,
        meta: { xp, usd },
        entries: [
          { accountId: bonusAcc, amount: -usd },
          { accountId: userAcc, amount: usd },
        ],
        idempotencyKey: `xp_cashout_${userId}_${Date.now()}`,
      });

      if (!ok.ok) {
        await client.query('ROLLBACK');
        return { ok: false, error: ok.error ?? 'ledger_failed' };
      }

      const upd = await client.query(
        `UPDATE user_xp
         SET claimed_xp = claimed_xp + $1, updated_at = NOW()
         WHERE user_id = $2
         RETURNING claimed_xp`,
        [xp, userId],
      );

      await client.query(
        `INSERT INTO xp_transactions (user_id, amount, source, meta)
         VALUES ($1, $2, 'claim', $3)`,
        [userId, -xp, JSON.stringify({ usd, level: lvl })],
      );

      await client.query('COMMIT');
      return {
        ok: true,
        usd,
        newClaimedXp: Number(upd.rows[0]?.claimed_xp ?? claimedXp + xp),
      };
    } catch (e: any) {
      await client.query('ROLLBACK');
      return { ok: false, error: e.message };
    } finally {
      client.release();
    }
  }
